```dart
// lib/core/services/auth_service.dart
class AuthService {
  /// Sign in dengan email & password (async operation)
  static Future<Map<String, dynamic>> signIn({
    required String email,
    required String password,
  }) async {
    try {
      final response = await Supabase.instance.client.auth.signInWithPassword(
        email: email,
        password: password,
      );

      // Check user status
      final userData = await Supabase.instance.client
          .from('users')
          .select('status, id_role, role(nama)')
          .eq('id_auth', response.user!.id)
          .single();

      if (userData['status'] != 'aktif') {
        await signOut();
        throw Exception('Akun tidak aktif');
      }

      return {'success': true, 'data': userData};
    } catch (e) {
      return {'success': false, 'error': e.toString()};
    }
  }
}

// lib/core/services/ml_vegetable_service.dart
class MLVegetableService {
  /// Predict vegetable quality (async HTTP request)
  Future<Map<String, dynamic>> predictQuality(File imageFile) async {
    try {
      final formData = FormData.fromMap({
        'image': await MultipartFile.fromFile(
          imageFile.path,
          filename: imageFile.path.split('/').last,
        ),
      });

      final response = await _dio.post(
        '$baseUrl/predict',
        data: formData,
        options: Options(headers: {'Content-Type': 'multipart/form-data'}),
      );

      if (response.statusCode == 200) {
        return {
          'success': true,
          'prediction': response.data['quality'],
          'confidence': response.data['confidence'],
        };
      }
    } catch (e) {
      return {'success': false, 'error': e.toString()};
    }
  }
}
```

```dart
// Supabase Realtime Stream
class KegiatanService {
  /// Watch kegiatan dengan realtime updates
  Stream<List<KegiatanModel>> watchKegiatan() {
    return Supabase.instance.client
        .from('kegiatan')
        .stream(primaryKey: ['id'])
        .order('tanggal_kegiatan', ascending: false)
        .map((data) {
          return data.map((json) => KegiatanModel.fromJson(json)).toList();
        });
  }
}

// lib/core/providers/kegiatan_provider.dart

// lib/core/services/supabase_error_interceptor.dart
// Stream error handling
class SupabaseErrorInterceptor {
  static Stream<T> handleStream<T>(
    Stream<T> stream, {
    Function(Object error)? onError,
  }) {
    return stream.handleError((error) {
```

```dart
class MarketplaceService {
  static Future<List<ProdukMarketplaceModel>> getAllProducts() async {
    try {
      final response = await Supabase.instance.client
          .from('produk_marketplace')
          .select('''
            *,
            toko:id_toko(*),
            kategori_produk:id_kategori_produk(*)
          ''')
          .eq('status', 'tersedia')
          .order('created_at', ascending: false);

      return (response as List)
          .map((json) => ProdukMarketplaceModel.fromJson(json))
          .toList();
    } catch (e) {
      throw Exception('Failed to load products: $e');
    }
  }

  // Supabase REST API - POST
  static Future<String> createProduct({
    required String nama,
    required double harga,
    required String deskripsi,
    required int idToko,
    required int idKategori,
  }) async {
    try {
      final response = await Supabase.instance.client
          .from('produk_marketplace')
          .insert({
            'nama_produk': nama,

            'deskripsi': deskripsi,
            'id_toko': idToko,
            'id_kategori_produk': idKategori,

// External API dengan Dio
class MLVegetableService {
  final Dio _dio = Dio();
  static const String productionUrl = 'https://modelannkel6-production.up.railway.app';

  MLVegetableService() {
    _dio.options.connectTimeout = const Duration(seconds: 30);
    _dio.options.receiveTimeout = const Duration(seconds: 30);
  }

  // GET Request
  Future<bool> checkHealth() async {
    try {
      final response = await _dio.get('$productionUrl/');
      return response.statusCode == 200;
    } catch (e) {
      return false;
    }
  }

  // POST Request with FormData
  Future<Map<String, dynamic>> predictQuality(File imageFile) async {
    try {
      final formData = FormData.fromMap({
        'image': await MultipartFile.fromFile(
          imageFile.path,
          filename: imageFile.path.split('/').last,
        ),
      });

      final response = await _dio.post(
        '$productionUrl/predict',
        data: formData,
        options: Options(
          headers: {'Content-Type': 'multipart/form-data'},
        ),
      );

      if (response.statusCode == 200) {
        return {
          'success': true,
          'quality': response.data['quality'],
          'confidence': response.data['confidence'] * 100,
        };
      }

      throw Exception('Prediction failed');
    } catch (e) {
      return {'success': false, 'error': e.toString()};
    }
  }
}
```
